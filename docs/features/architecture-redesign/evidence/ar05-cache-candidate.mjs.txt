import {
  closeSync,
  fchmodSync,
  lstatSync,
  mkdirSync,
  openSync,
  readFileSync,
  renameSync,
  unlinkSync,
  writeFileSync,
} from "node:fs";
import { createHash, randomUUID } from "node:crypto";
import { tmpdir } from "node:os";
import { fileURLToPath } from "node:url";
import { join, resolve } from "node:path";

const root = fileURLToPath(new URL("../", import.meta.url));
const hookMode = process.argv.includes("--hook");
const target = process.argv.find((argument, index) => index > 1 && argument !== "--hook");
const normalize = (value) => value.trim().replace(/\s+/g, " ");
const designFiles = ["colors", "typography", "spacing"].map((name) =>
  resolve(root, `docs/design-system/${name}.md`),
);
const filename = target ? resolve(target) : resolve(root, "docs/architecture/index.html");

// The cache only shortens an optional hook invocation. Every mandatory check parses the inputs.
function successfulSnapshotCache() {
  if (process.platform === "win32" || process.argv.slice(2).join("\0") !== "--hook") return null;
  const inputs = [...designFiles, filename];
  try {
    if (inputs.some((path) => !lstatSync(path).isFile() || lstatSync(path).isSymbolicLink()))
      return null;
    const bytes = inputs.map((path) => readFileSync(path));
    const checker = readFileSync(fileURLToPath(import.meta.url));
    const rootIdentity = resolve(root);
    const rootHash = createHash("sha256").update(rootIdentity).digest("hex");
    const signatureHash = createHash("sha256")
      .update("design-hook-cache-v1\0")
      .update(rootIdentity);
    for (const value of [checker, ...bytes]) {
      signatureHash.update(String(value.length)).update("\0").update(value);
    }
    const signature = signatureHash.digest("hex");
    const directory = join(tmpdir(), `cc-harness-design-${rootHash}`);
    const cacheFile = join(directory, "success.json");
    const privateFile = (path, mode) => {
      const stat = lstatSync(path);
      return (
        !stat.isSymbolicLink() &&
        (mode === 0o700 ? stat.isDirectory() : stat.isFile()) &&
        (stat.mode & 0o777) === mode &&
        (typeof process.getuid !== "function" || stat.uid === process.getuid())
      );
    };
    const cache = {
      texts: bytes.map((value) => value.toString("utf8")),
      hit() {
        try {
          if (!privateFile(directory, 0o700) || !privateFile(cacheFile, 0o600)) return false;
          const record = JSON.parse(readFileSync(cacheFile, "utf8"));
          return record.version === 1 && record.signature === signature;
        } catch {
          return false;
        }
      },
      record() {
        let temporary;
        try {
          try {
            mkdirSync(directory, { mode: 0o700 });
          } catch (error) {
            if (error.code !== "EEXIST") return;
          }
          if (!privateFile(directory, 0o700)) return;
          try {
            if (!privateFile(cacheFile, 0o600)) return;
          } catch (error) {
            if (error.code !== "ENOENT") return;
          }
          temporary = join(directory, `.success-${randomUUID()}.tmp`);
          const descriptor = openSync(temporary, "wx", 0o600);
          try {
            fchmodSync(descriptor, 0o600);
            writeFileSync(descriptor, JSON.stringify({ version: 1, signature }) + "\n");
          } finally {
            closeSync(descriptor);
          }
          // A replaced or linked destination is never a reason to skip validation.
          try {
            if (!privateFile(cacheFile, 0o600)) return;
          } catch (error) {
            if (error.code !== "ENOENT") return;
          }
          renameSync(temporary, cacheFile);
          temporary = undefined;
        } catch {
          // Cache failures never change the validator's result.
        } finally {
          if (temporary) {
            try {
              unlinkSync(temporary);
            } catch {
              // A failed cleanup does not change the validator's result.
            }
          }
        }
      },
    };
    return cache;
  } catch {
    return null;
  }
}

try {
  const cache = successfulSnapshotCache();
  if (cache?.hit()) process.exit(0);
  const rules = {};
  for (const [index, name] of ["colors", "typography", "spacing"].entries()) {
    const markdown = cache?.texts[index] ?? readFileSync(designFiles[index], "utf8");
    const block = markdown.match(/```json\s*([\s\S]*?)```/);
    if (!block) throw new Error(`${name}.md: JSON 기준이 없습니다.`);
    for (const [property, values] of Object.entries(JSON.parse(block[1]))) {
      if (
        rules[property] ||
        !Array.isArray(values) ||
        !values.length ||
        values.some((value) => typeof value !== "string")
      ) {
        throw new Error(`${name}.md: 잘못되거나 중복된 속성 ${property}`);
      }
      rules[property] = new Set(values.map(normalize));
    }
  }
  const html = cache?.texts[3] ?? readFileSync(filename, "utf8");
  if (/\sstyle\s*=|<link\b[^>]*\bstylesheet\b/i.test(html)) {
    throw new Error(
      "인라인 속성·외부 스타일은 현재 검사 범위 밖입니다. 검사 범위를 먼저 확장하세요.",
    );
  }
  const styles = [...html.matchAll(/<style\b[^>]*>([\s\S]*?)<\/style>/gi)];
  if (!styles.length) throw new Error("검사할 style 블록이 없습니다.");
  const violations = [];
  let checked = 0;
  for (const [, source] of styles) {
    const css = source.replace(/\/\*[\s\S]*?\*\//g, "");
    for (const [, property, rawValue] of css.matchAll(/([\w-]+)\s*:\s*([^;{}]+)(?:;|(?=\}))/g)) {
      const value = normalize(rawValue);
      checked += 1;
      if (!rules[property]?.has(value)) violations.push(`${property}: ${value}`);
    }
  }
  if (!checked) throw new Error("CSS 선언이 없어 검사하지 못했습니다.");
  if (violations.length)
    throw new Error(`디자인 기준 이탈 (${filename}):\n${violations.join("\n")}`);
  cache?.record();
  console.log(hookMode ? "{}" : `디자인 검사 통과: ${checked}개 선언 (${filename})`);
} catch (error) {
  console.error(
    `${error.message}\n기준 문서와 변경 의도를 비교해 수정하세요. 자동 복구는 하지 않습니다.`,
  );
  process.exitCode = hookMode ? 2 : 1;
}
